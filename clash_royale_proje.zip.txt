// ---- Kart Görsel Yükleme Hook'u ----
const useCardImages = (cards) => {
  const [images, setImages] = useState({});
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  useEffect(() => {
    const loadImages = async () => {
      setLoading(true);
      setError(null);
      
      try {
        // Yöntem 1: Kendi API'niz üzerinden proxy (ÖNERİLEN)
        const response = await fetch('/api/clash-royale/cards', {
          headers: {
            'Authorization': `Bearer ${CLASH_ROYALE_API.API_TOKEN}`,
            'Content-Type': 'application/json'
          }
        });
        
        if (!response.ok) {
          throw new Error('API yanıt vermedi - yerel görsellere geçiliyor');
        }
        
        const data = await response.json();
        
        // Görselleri eşleştir
        const imageMap = {};
        data.items?.forEach(apiCard => {
          const localCard = cards.find(c => 
            c.name.toLowerCase() === apiCard.name.toLowerCase()
          );
          if (localCard && apiCard.iconUrls?.medium) {
            imageMap[localCard.id] = apiCard.iconUrls.medium;
          }
        });
        
        setImages(imageMap);
      } catch (err) {
        console.log('API yüklenemedi, yerel görseller kullanılıyor:', err.message);
        setError(err.message);
        
        // Yedek: Yerel görselleri yükle
        const localImages = {};
        cards.forEach(card => {
          if (card.localImage) {
            localImages[card.id] = card.localImage;
          } else if (card.placeholder) {
            localImages[card.id] = card.placeholder;
          }
        });
        setImages(localImages);
      } finally {
        setLoading(false);
      }
    };

    if (cards.length > 0) {
      loadImages();
    }
  }, [cards]);

  return { images, loading, error };
};

// ---- Telifsiz Kart Görsel Bileşeni ----
function CardImage({ card, size = "medium" }) {
  const [imgError, setImgError] = useState(false);
  const [isLoaded, setIsLoaded] = useState(false);
  
  const sizeClasses = {
    small: "w-8 h-8",
    medium: "w-12 h-12",
    large: "w-16 h-16"
  };

  // Görsel URL'sini belirle (öncelik sırası: API > Yerel > Placeholder)
  const imageUrl = card.imageUrl || card.localImage || card.placeholder;
  
  if (!imageUrl || imgError) {
    // Görsel yoksa veya yükleme hatası varsa - minimalist ikon göster
    return (
      <div className={`${sizeClasses[size]} bg-gray-700 rounded-lg flex items-center justify-center`}>
        <div className="text-center">
          <div className="text-xs font-bold text-white">{card.elixir}</div>
          <div className="text-[8px] text-gray-400">💧</div>
        </div>
      </div>
    );
  }

  return (
    <div className={`${sizeClasses[size]} relative`}>
      {!isLoaded && (
        <div className="absolute inset-0 bg-gray-700 rounded-lg animate-pulse flex items-center justify-center">
          <Loader2 className="w-4 h-4 text-gray-400 animate-spin" />
        </div>
      )}
      <img
        src={imageUrl}
        alt={card.name}
        className={`${sizeClasses[size]} object-cover rounded-lg transition-opacity duration-300 ${
          isLoaded ? 'opacity-100' : 'opacity-0'
        }`}
        onLoad={() => setIsLoaded(true)}
        onError={() => setImgError(true)}
        loading="lazy"
      />
    </div>
  );
}

              onClick={() => onRemove(card.id)}

            </p>
          </div>
        )}
      </div>
    </div>
  );
}

// ---- Placeholder Üretici (Görsel kullanmak istemezseniz) ----
function CardPlaceholder({ card }) {
  const bgColor = ROLE_COLORS[card.role] || "#8a8fa3";
  const initials = card.name.split(' ').map(w => w[0]).join('').toUpperCase();
  
  return (
    <div 
      className="w-full h-full rounded-lg flex flex-col items-center justify-center"
      style={{ 
        background: `linear-gradient(135deg, ${bgColor}80, ${bgColor}40)`,
        borderColor: bgColor,
        borderWidth: '2px'
      }}
    >
      <span className="text-2xl font-bold text-white">{initials}</span>
      <span className="text-xs text-white/70 mt-1">{card.elixir} 💧</span>
    </div>
  );
}
